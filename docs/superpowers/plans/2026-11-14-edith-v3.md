# EDITH v3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the single-pass keyword `main.py` with a continuous Windows voice assistant: offline wake word, offline speech-to-text, an LLM brain (FreeLLMAPI with a local Qwen fallback), and pluggable skills.

**Architecture:** `voice/` turns audio into text and back. `brain/llm.py` hides which LLM backend answered and always returns a `Reply`. `brain/agent.py` runs the chat-plus-tool loop over a skill registry built by the `@skill` decorator in `skills/__init__.py`. `main.py` wires them together: wait for wake word, converse, return to idle.

**Tech Stack:** Python 3.12 (Windows), `openai` SDK, `llama-cpp-python` (CPU wheel), `huggingface_hub`, `vosk`, `faster-whisper`, `sounddevice`, `numpy`, `pyttsx3`, `pyautogui`, `psutil`, `requests`, `python-dotenv`, `pytest`.

**Spec:** `docs/superpowers/specs/2026-11-14-edith-v3-design.md`

**Shell:** all commands are for Git Bash on Windows, run from `d:\edith` (`/d/edith`).

```python
import logging
import os
from dataclasses import dataclass
from pathlib import Path

from dotenv import load_dotenv

ROOT = Path(__file__).resolve().parent


@dataclass(frozen=True)
class Config:
    base_url: str | None
    api_key: str | None
    model: str
    mic_threshold: float
    wake_word: str
    data_dir: Path
    models_dir: Path
```

```python
import inspect
import typing
from dataclasses import dataclass
from typing import Callable

_JSON_TYPES = {str: "string", int: "integer", float: "number", bool: "boolean"}


@dataclass
class Skill:
    func: Callable[..., str]
    confirm: bool
    schema: dict


REGISTRY: dict[str, Skill] = {}


def _schema(func: Callable) -> dict:
    hints = typing.get_type_hints(func)
    properties, required = {}, []
    for name, param in inspect.signature(func).parameters.items():
        properties[name] = {"type": _JSON_TYPES[hints.get(name, str)]}
        if param.default is inspect.Parameter.empty:
            required.append(name)
    return {
        "type": "function",
        "function": {
            "name": func.__name__,
            "description": inspect.getdoc(func) or "",
            "parameters": {"type": "object", "properties": properties, "required": required},
        },
    }


def skill(confirm: bool = False):
    """Register a function as a skill the LLM can call."""

    def wrap(func):
        REGISTRY[func.__name__] = Skill(func, confirm, _schema(func))
        return func

    return wrap


def tool_schemas() -> list[dict]:
    return [s.schema for s in REGISTRY.values()]


def load_all() -> None:
    # Importing each module runs its @skill decorators.
    from skills import info, media, productivity, system, web  # noqa: F401
```

```python
import json
import logging
from dataclasses import dataclass, field

log = logging.getLogger(__name__)

LOCAL_TOOL_PROMPT = """You can use these tools:
{tools}
To use a tool, reply with only a JSON object like {{"tool": "tool_name", "args": {{"name": "value"}}}} and nothing else.
After you get the tool result, answer the user in plain words."""


@dataclass
class Reply:
    text: str | None
    tool_calls: list[dict] = field(default_factory=list)  # [{"id", "name", "arguments": dict}]
    message: dict = field(default_factory=dict)  # assistant message to append to history as-is


def _loads_args(raw: str | None) -> dict:
    try:
        args = json.loads(raw or "{}")
    except ValueError:
        return {}
    return args if isinstance(args, dict) else {}


def parse_local_reply(text: str) -> Reply:
    plain = Reply(text, [], {"role": "assistant", "content": text})
    start, end = text.find("{"), text.rfind("}")
    if start == -1 or end < start:
        return plain
    try:
        data = json.loads(text[start : end + 1])
    except ValueError:
        return plain
    if not isinstance(data, dict) or not isinstance(data.get("tool"), str):
        return plain
    args = data.get("args") if isinstance(data.get("args"), dict) else {}
    call = {"id": "local-1", "name": data["tool"], "arguments": args}
    message = {
        "role": "assistant",
        "content": None,
        "tool_calls": [
            {"id": "local-1", "type": "function", "function": {"name": data["tool"], "arguments": json.dumps(args)}}
        ],
    }
    return Reply(None, [call], message)


def to_local_messages(messages: list[dict], tools: list[dict]) -> list[dict]:
    """Rewrite OpenAI tool turns as plain chat turns the local model understands."""
    tool_lines = "\n".join(
        f"- {t['function']['name']}: {t['function']['description']} "
        f"args: {json.dumps(t['function']['parameters']['properties'])}"
        for t in tools
    )
    out = []
    for m in messages:
        role = m["role"]
        if role == "system":
            extra = "\n\n" + LOCAL_TOOL_PROMPT.format(tools=tool_lines) if tools else ""
            out.append({"role": "system", "content": m["content"] + extra})
        elif role == "tool":
            out.append({"role": "user", "content": f"Tool result: {m['content']}"})
        elif role == "assistant" and m.get("tool_calls"):
            fn = m["tool_calls"][0]["function"]
            out.append(
                {"role": "assistant", "content": json.dumps({"tool": fn["name"], "args": _loads_args(fn["arguments"])})}
            )
        else:
            out.append({"role": role, "content": m.get("content") or ""})
    return out
```

```python
REMOTE_TIMEOUT = 8.0  # seconds
RETRY_REMOTE_AFTER = 60.0  # seconds offline before trying FreeLLMAPI again
REMOTE_ERRORS = (openai.APIConnectionError, openai.RateLimitError, openai.InternalServerError)

LOCAL_REPO = "Qwen/Qwen2.5-3B-Instruct-GGUF"
LOCAL_FILE = "qwen2.5-3b-instruct-q4_k_m.gguf"


class LLMUnavailable(Exception):
    """Neither FreeLLMAPI nor the local model could answer."""


class LLM:
    def __init__(self, remote, model: str, local_factory: Callable, on_fallback: Callable | None = None,
                 clock: Callable[[], float] = time.monotonic):
        self.remote = remote
        self.model = model
        self.local_factory = local_factory
        self.on_fallback = on_fallback
        self.clock = clock
        self._local = None
        self._offline_until = 0.0
        self._announced = False

    def chat(self, messages: list[dict], tools: list[dict]) -> Reply:
        if self.remote is not None and self.clock() >= self._offline_until:
            try:
                reply = self._remote_chat(messages, tools)
                self._announced = False
                return reply
            except REMOTE_ERRORS as e:
                log.warning("FreeLLMAPI failed, using local model: %s", e)
                self._offline_until = self.clock() + RETRY_REMOTE_AFTER
                if self.on_fallback and not self._announced:
                    self._announced = True
                    self.on_fallback()
        try:
            if self._local is None:
                self._local = self.local_factory()
            return parse_local_reply(self._local.complete(to_local_messages(messages, tools)))
        except Exception as e:
            log.exception("local model failed")
            raise LLMUnavailable(str(e)) from e

    def _remote_chat(self, messages: list[dict], tools: list[dict]) -> Reply:
        extra = {"tools": tools} if tools else {}
        resp = self.remote.chat.completions.create(
            model=self.model, messages=messages, timeout=REMOTE_TIMEOUT, **extra
        )
        msg = resp.choices[0].message
        calls = [
            {"id": tc.id, "name": tc.function.name, "arguments": _loads_args(tc.function.arguments)}
            for tc in msg.tool_calls or []
        ]
        return Reply(msg.content, calls, msg.model_dump(exclude_none=True))


class LocalModel:
    """Qwen2.5-3B on CPU through llama-cpp-python."""

    def __init__(self, model_path: Path):
        from llama_cpp import Llama

        self.llm = Llama(model_path=str(model_path), n_ctx=4096, n_threads=4, verbose=False)

    def complete(self, messages: list[dict]) -> str:
        out = self.llm.create_chat_completion(messages=messages, max_tokens=256, temperature=0.3)
        return out["choices"][0]["message"]["content"] or ""


def download_local_model(models_dir: Path) -> Path:
    from huggingface_hub import hf_hub_download

    return Path(hf_hub_download(repo_id=LOCAL_REPO, filename=LOCAL_FILE, local_dir=models_dir))


def build_llm(cfg, on_fallback: Callable | None = None) -> LLM:
    remote = None
    if cfg.base_url and cfg.api_key:
        remote = openai.OpenAI(base_url=cfg.base_url.rstrip("/") + "/v1", api_key=cfg.api_key, max_retries=0)
    return LLM(
        remote,
        cfg.model,
        local_factory=lambda: LocalModel(download_local_model(cfg.models_dir)),
        on_fallback=on_fallback,
    )
```

Notes:
- `openai.APITimeoutError` is a subclass of `APIConnectionError`, so timeouts are covered.
- `max_retries=0` makes the fallback fast. The SDK default retries twice, which would triple the wait.

```python
import logging
from datetime import datetime
from typing import Callable

from brain.llm import LLMUnavailable
from skills import REGISTRY, Skill

log = logging.getLogger(__name__)

SYSTEM_PROMPT = (
    "You are EDITH, a voice assistant running on the user's Windows PC. Address the user as sir. "
    "Your replies are spoken aloud: answer in one to three short sentences, with no markdown, lists or emojis. "
    "Use the tools to act on the PC or fetch information. Current date and time: {now}."
)


class Agent:
    def __init__(self, llm, confirm: Callable[[str], bool], registry: dict[str, Skill] = REGISTRY,
                 max_tool_calls: int = 3, max_turns: int = 10):
        self.llm = llm
        self.confirm = confirm  # asks the user out loud, returns True only on "yes"
        self.registry = registry
        self.max_tool_calls = max_tool_calls
        self.max_turns = max_turns
        self.history: list[dict] = []

    def reset(self) -> None:
        self.history = []

    def handle(self, text: str) -> str:
        start = len(self.history)
        self.history.append({"role": "user", "content": text})
        try:
            answer = self._loop()
        except LLMUnavailable:
            del self.history[start:]
            return "My brain is offline right now, sir."
        except Exception:
            log.exception("agent failed")
            del self.history[start:]
            return "Sorry sir, something went wrong."
        self._trim()
        return answer

    def _loop(self) -> str:
        tools = [s.schema for s in self.registry.values()]
        calls = 0
        while calls < self.max_tool_calls:
            reply = self.llm.chat(self._messages(), tools)
            self.history.append(reply.message)
            if not reply.tool_calls:
                return (reply.text or "").strip()
            for call in reply.tool_calls:
                calls += 1
                # Every tool call needs a tool message, even past the cap.
                result = self._run(call) if calls <= self.max_tool_calls else "error: tool call limit reached"
                self.history.append({"role": "tool", "tool_call_id": call["id"], "content": result})
        reply = self.llm.chat(self._messages(), [])  # no tools: force a spoken answer
        self.history.append(reply.message)
        return (reply.text or "").strip()

    def _run(self, call: dict) -> str:
        entry = self.registry.get(call["name"])
        if entry is None:
            return f"error: unknown tool {call['name']}"
        if entry.confirm and not self.confirm(call["name"].replace("_", " ")):
            return "cancelled by user"
        try:
            return str(entry.func(**call["arguments"]))
        except Exception as e:
            log.exception("skill %s failed", call["name"])
            return f"error: {e}"

    def _messages(self) -> list[dict]:
        system = SYSTEM_PROMPT.format(now=datetime.now().strftime("%A %d %B %Y, %I:%M %p"))
        return [{"role": "system", "content": system}] + self.history

    def _trim(self) -> None:
        user_idx = [i for i, m in enumerate(self.history) if m["role"] == "user"]
        if len(user_idx) > self.max_turns:
            # Cut at a user message so tool calls and their results stay paired.
            self.history = self.history[user_idx[-self.max_turns]:]
```

```python
import logging
import queue
import threading

log = logging.getLogger(__name__)


class Speaker:
    """pyttsx3 is not thread-safe, so one thread owns the engine and all speech goes through a queue."""

    def __init__(self):
        self._queue: queue.Queue = queue.Queue()
        threading.Thread(target=self._run, daemon=True, name="tts").start()

    def say(self, text: str, wait: bool = True) -> None:
        print(f"EDITH: {text}")
        done = threading.Event()
        self._queue.put((text, done))
        if wait:
            done.wait()

    def _run(self) -> None:
        import comtypes  # installed with pyttsx3 on Windows; SAPI needs COM initialised on this thread
        import pyttsx3

        comtypes.CoInitialize()
        engine = pyttsx3.init()
        while True:
            text, done = self._queue.get()
            try:
                engine.say(text)
                engine.runAndWait()
            except Exception:
                log.exception("speech failed")
            finally:
                done.set()
```

```python
import queue
from pathlib import Path
from typing import Iterator

import numpy as np

SAMPLE_RATE = 16000
CHUNK_SEC = 0.1
EPS = 1e-9  # repeated 0.1 additions are not exact in floating point


def collect_speech(chunks: Iterator[np.ndarray], chunk_sec: float, threshold: float,
                   start_timeout: float, silence: float, max_len: float) -> np.ndarray | None:
    """Return audio from the first loud chunk until `silence` seconds of quiet, or None if nobody spoke."""
    frames: list[np.ndarray] = []
    waited = quiet = 0.0
    for chunk in chunks:
        loud = float(np.sqrt(np.mean(chunk ** 2))) > threshold
        if not frames:
            if not loud:
                waited += chunk_sec
                if waited + EPS >= start_timeout:
                    return None
                continue
        frames.append(chunk)
        quiet = 0.0 if loud else quiet + chunk_sec
        if quiet + EPS >= silence or len(frames) * chunk_sec + EPS >= max_len:
            break
    return np.concatenate(frames) if frames else None


def mic_chunks(chunk_sec: float = CHUNK_SEC) -> Iterator[np.ndarray]:
    import sounddevice as sd

    q: queue.Queue = queue.Queue()
    with sd.InputStream(samplerate=SAMPLE_RATE, channels=1, dtype="float32",
                        blocksize=int(SAMPLE_RATE * chunk_sec), callback=lambda data, *_: q.put(data[:, 0].copy())):
        while True:
            yield q.get()


class Listener:
    def __init__(self, models_dir: Path, threshold: float = 0.01, model_name: str = "base.en"):
        from faster_whisper import WhisperModel

        self.threshold = threshold
        self.model = WhisperModel(model_name, device="cpu", compute_type="int8",
                                  download_root=str(models_dir / "whisper"))

    def listen(self, start_timeout: float = 10.0, silence: float = 1.0, max_len: float = 15.0) -> str | None:
        """Return the spoken text, "" if speech was unclear, or None if nobody spoke before start_timeout."""
        chunks = mic_chunks()
        try:
            audio = collect_speech(chunks, CHUNK_SEC, self.threshold, start_timeout, silence, max_len)
        finally:
            chunks.close()  # closes the microphone stream
        if audio is None:
            return None
        segments, _ = self.model.transcribe(audio, language="en", beam_size=1, vad_filter=True)
        return " ".join(s.text.strip() for s in segments).strip()
```

```python
import json
import logging
import queue
import urllib.request
import zipfile
from pathlib import Path

log = logging.getLogger(__name__)

SAMPLE_RATE = 16000
VOSK_NAME = "vosk-model-small-en-us-0.15"
VOSK_URL = f"https://alphacephei.com/vosk/models/{VOSK_NAME}.zip"


def heard(text: str, word: str) -> bool:
    return word in text.split()


def ensure_vosk_model(models_dir: Path) -> Path:
    target = models_dir / VOSK_NAME
    if target.is_dir():
        return target
    archive = models_dir / f"{VOSK_NAME}.zip"
    log.info("downloading %s", VOSK_URL)
    urllib.request.urlretrieve(VOSK_URL, archive)
    with zipfile.ZipFile(archive) as z:
        z.extractall(models_dir)
    archive.unlink()
    return target


class WakeWord:
    def __init__(self, model_path: Path, word: str = "edith"):
        from vosk import Model, SetLogLevel

        SetLogLevel(-1)
        self.model = Model(str(model_path))
        self.word = word

    def wait(self) -> None:
        """Block until the wake word is heard."""
        import sounddevice as sd
        from vosk import KaldiRecognizer

        rec = KaldiRecognizer(self.model, SAMPLE_RATE, json.dumps([self.word, "[unk]"]))
        q: queue.Queue = queue.Queue()
        with sd.RawInputStream(samplerate=SAMPLE_RATE, blocksize=4000, dtype="int16", channels=1,
                               callback=lambda data, *_: q.put(bytes(data))):
            while True:
                data = q.get()
                if rec.AcceptWaveform(data):
                    text = json.loads(rec.Result())["text"]
                else:
                    text = json.loads(rec.PartialResult())["partial"]
                if heard(text, self.word):
                    return
```

```python
import logging
import re
import time

log = logging.getLogger("edith")

GOODBYE_WORDS = {"bye", "goodbye", "stop"}


def is_goodbye(text: str) -> bool:
    words = re.findall(r"[a-z]+", text.lower())
    # Short utterances only, so "stop the music on youtube" still reaches the agent.
    return len(words) <= 3 and bool(GOODBYE_WORDS & set(words))


def converse(agent, listener, speaker) -> None:
    """Talk without the wake word until silence, goodbye, or a microphone error."""
    while True:
        try:
            text = listener.listen(start_timeout=10.0)
        except Exception:
            log.exception("listening failed")
            speaker.say("I can't hear you, sir. Please check the microphone.")
            return
        if text is None:
            return
        if not text:
            speaker.say("I didn't catch that, sir.")
            continue
        print(f"You: {text}")
        if is_goodbye(text):
            speaker.say("Goodbye sir.")
            return
        speaker.say(agent.handle(text))


def main() -> None:
    import skills
    from brain.agent import Agent
    from brain.llm import build_llm
    from config import load_config, setup_logging
    from skills import productivity
    from voice.stt import Listener
    from voice.tts import Speaker
    from voice.wake import WakeWord, ensure_vosk_model

    cfg = load_config()
    setup_logging(cfg.data_dir)
    speaker = Speaker()
    speaker.say("Edith is starting up, sir. Please wait.")

    skills.load_all()
    productivity.announce = lambda text: speaker.say(text, wait=False)
    listener = Listener(cfg.models_dir, cfg.mic_threshold)
    wake = WakeWord(ensure_vosk_model(cfg.models_dir), cfg.wake_word)
    llm = build_llm(cfg, on_fallback=lambda: speaker.say("Switching to offline mode, sir."))

    def confirm(action: str) -> bool:
        speaker.say(f"Are you sure you want to {action}?")
        answer = listener.listen(start_timeout=6.0) or ""
        return "yes" in answer.lower()

    agent = Agent(llm, confirm)
    speaker.say("Edith 3 point o is online. Say Edith when you need me.")

    while True:
        try:
            wake.wait()
        except Exception:
            log.exception("wake word failed")
            speaker.say("I can't hear you, sir. Please check the microphone.")
            time.sleep(5)
            continue
        speaker.say("Yes sir?")
        converse(agent, listener, speaker)
        agent.reset()


if __name__ == "__main__":
    main()
```

````markdown
A Windows voice assistant. Say "Edith", then talk normally. EDITH answers with an LLM and can open websites and apps, play YouTube, check the weather, read Wikipedia, control volume, lock or shut down the PC, set reminders and take notes.
````
